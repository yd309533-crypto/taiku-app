    <script>
        document.addEventListener('DOMContentLoaded', () => {
            const reflectionInput = document.getElementById('reflection');
            const submitBtn = document.getElementById('submit-btn');
            const resetBtn = document.getElementById('reset-btn');
            const inputSection = document.getElementById('input-section');
            const loadingSection = document.getElementById('loading-section');
            const resultSection = document.getElementById('result-section');
            const starsContainer = document.getElementById('stars-container');
            const scoreText = document.getElementById('score-text');
            const feedbackText = document.getElementById('feedback-text');
            const errorMessage = document.getElementById('error-message');
            const submittedReflectionText = document.getElementById('submitted-reflection-text');

            // APIの呼び出しと評価の実行
            async function evaluateReflection(text) {
                // Canvas環境下では空文字にしておくことで自動的にAPIキーが注入されます
            constapiKey="AQ.Ab8RN6LkylgqKoliIM9vARjqgcK_H2im9-L4YU_jhl2Lhbadqw"; 
                const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${apiKey}`;

                const systemPrompt = `
あなたは小学校の温かく熱心な体育の先生です。児童が書いた体育の授業の振り返りを読み、文部科学省の「小学校学習指導要領（体育編）」が示す3つの目標・観点（1. 知識及び技能、 2. 思考力、判断力、表現力等、 3. 学びに向かう力、人間性等）に基づき、5段階で評価し、フィードバックを行ってください。

【対象】
小学校高学年（5・6年生）

【評価の目安】
5点: 3つの観点（できたこと、工夫したこと、次への意欲や友達との関わりなど）がとても具体的に書かれている。自らの課題を見つけ、解決に向けた論理的な思考が見られる。
4点: 複数の観点が具体的に書かれている。
3点: 授業の様子や感想が素直に書かれている。
2点: 感想のみで、具体的な学びの内容が少ない。
1点: 記述が極端に短い、または体育に全く関係がない。

【フィードバックの書き方（超重要）】
・小学校高学年向けに、適切な漢字を使用してください。
・児童の頑張りや気づきを褒めるだけでなく、なぜそれが良かったのかを具体的に示し、さらに深い学びや他者との協働に繋がるような言葉がけをしてください。
・ポジティブなトーンを維持しつつ、次のステップに向けた具体的なアドバイスを含めてください。

出力は必ず以下のJSONスキーマに従ってください。
`;

                const payload = {
                    contents: [{ parts: [{ text: text }] }],
                    systemInstruction: { parts: [{ text: systemPrompt }] },
                    generationConfig: {
                        responseMimeType: "application/json",
                        responseSchema: {
                            type: "OBJECT",
                            properties: {
                                score: { 
                                    type: "INTEGER", 
                                    description: "1から5までの5段階評価の数値" 
                                },
                                feedback: { 
                                    type: "STRING", 
                                    description: "小学校高学年に向けたフィードバックコメント" 
                                }
                            },
                            required: ["score", "feedback"]
                        }
                    }
                };

                try {
                    const response = await fetch(apiUrl, {
                        method: 'POST',
                        headers: { 'Content-Type': 'application/json' },
                        body: JSON.stringify(payload)
                    });

                    if (!response.ok) {
                        throw new Error(`HTTP error! status: ${response.status}`);
                    }

                    const result = await response.json();
                    
                    if (result.candidates && result.candidates.length > 0 &&
                        result.candidates[0].content && result.candidates[0].content.parts &&
                        result.candidates[0].content.parts.length > 0) {
                        
                        const jsonText = result.candidates[0].content.parts[0].text;
                        return JSON.parse(jsonText);
                    } else {
                        throw new Error("APIからの応答形式が予期せぬものでした。");
                    }
                } catch (error) {
                    console.error("評価エラー:", error);
                    throw error;
                }
            }

            // 星を描画する関数
            function renderStars(score) {
                starsContainer.innerHTML = '';
                for (let i = 1; i <= 5; i++) {
                    const star = document.createElement('span');
                    star.textContent = '★';
                    if (i <= score) {
                        star.className = 'star-active filter drop-shadow-md';
                    } else {
                        star.className = 'star-inactive';
                    }
                    starsContainer.appendChild(star);
                }
            }

            // 送信ボタンのクリックイベント
            submitBtn.addEventListener('click', async () => {
                const text = reflectionInput.value.trim();
                
                if (!text) {
                    errorMessage.textContent = '振り返りを書いてからボタンを押してください。';
                    errorMessage.classList.remove('hidden');
                    return;
                }
                
                if (text.length < 5) {
                    errorMessage.textContent = 'もう少し詳しく書いてみましょう。';
                    errorMessage.classList.remove('hidden');
                    return;
                }

                // UIの切り替え
                errorMessage.classList.add('hidden');
                inputSection.classList.add('hidden');
                loadingSection.classList.remove('hidden');

                try {
                    // APIコール
                    const evaluation = await evaluateReflection(text);
                    
                    // 結果の反映
                    renderStars(evaluation.score);
                    
                    // スコアに応じたコメント
                    let scoreComment = "";
                    if (evaluation.score === 5) scoreComment = "大変素晴らしい！！";
                    else if (evaluation.score === 4) scoreComment = "とても良いですね！";
                    else if (evaluation.score === 3) scoreComment = "よく書けました！";
                    else if (evaluation.score === 2) scoreComment = "頑張って書きましたね！";
                    else scoreComment = "書いてくれてありがとう！";
                    
                    scoreText.textContent = `${evaluation.score}点！ ${scoreComment}`;
                    feedbackText.textContent = evaluation.feedback;
                    submittedReflectionText.textContent = text;

                    // UIの切り替え
                    loadingSection.classList.add('hidden');
                    resultSection.classList.remove('hidden');

                } catch (error) {
                    // エラー時の処理
                    loadingSection.classList.add('hidden');
                    inputSection.classList.remove('hidden');
                    errorMessage.textContent = '通信エラーが発生しました。もう一度試してください。';
                    errorMessage.classList.remove('hidden');
                }
            });

            // リセットボタンのクリックイベント
            resetBtn.addEventListener('click', () => {
                reflectionInput.value = '';
                resultSection.classList.add('hidden');
                inputSection.classList.remove('hidden');
            });
        });
    </script>
